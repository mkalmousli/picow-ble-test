cmake_minimum_required(VERSION 3.13)
set(PICO_BOARD pico_w)
include($ENV{PICO_SDK_PATH}/external/pico_sdk_import.cmake)
project(bletest C CXX ASM)
pico_sdk_init()
add_executable(bletest main.c)
target_include_directories(bletest PRIVATE ${CMAKE_CURRENT_LIST_DIR})
target_link_libraries(bletest pico_stdlib pico_cyw43_arch_none pico_btstack_ble pico_btstack_cyw43)
pico_enable_stdio_usb(bletest 1)
pico_add_extra_outputs(bletest)
